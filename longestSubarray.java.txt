//Longest Subarray With Sum K
//Last Updated : 11 Jan, 2025
//Given an array arr[] of size n containing integers, the task is to find the length of the longest subarray //having sum equal to the given value k.

//Note: If there is no subarray with sum equal to k, return 0.

import java.util.HashMap;
class Main {
    public static int longestSubarray(int [] arr,int k){
        int max=-1,prefixSum=0;
        HashMap<Integer,Integer> map=new HashMap<>();
        
        for(int i=0;i<arr.length;i++){
            prefixSum+=arr[i];
            if(k==prefixSum){
                max=i+1;
            }
            if(map.containsKey(prefixSum-k)){
               max=Math.max(max,i-map.get(prefixSum-k)+1);
            }
            else if(!map.containsKey(prefixSum)){
                map.put(prefixSum,i+1);
            }
        }
        return max;
    }
    public static void main(String[] args) {
        System.out.println(longestSubarray(new int[]{2,1,4,-5,1,1,1,1,-1,5},3));
    }
}
